cmake_minimum_required(VERSION 3.18)
project(urnetwork_cpp_embed_server LANGUAGES CXX)
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(embed-server embed_server.cpp)
target_compile_features(embed-server PRIVATE cxx_std_17)
target_link_libraries(embed-server PRIVATE CURL::libcurl nlohmann_json::nlohmann_json)
enable_testing()
add_test(NAME embed-server-self-test COMMAND embed-server --self-test)
