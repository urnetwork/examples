cmake_minimum_required(VERSION 3.24)
project(urnetwork_socket_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(urnetwork-sdk CONFIG REQUIRED)
find_package(Boost REQUIRED)
add_executable(socket-example main.cpp)
target_link_libraries(socket-example PRIVATE urnetwork::sdk Boost::headers)
option(ENABLE_CPR "Build the CPR HTTP proxy example" OFF)
if(ENABLE_CPR)
  find_package(cpr CONFIG REQUIRED)
  add_executable(cpr-example cpr_example.cpp)
  target_link_libraries(cpr-example PRIVATE cpr::cpr)
endif()
