# Builds the C++ provider example and its self-test, which does not link the
# SDK library, with the SDK package's urnetwork::sdk target and nlohmann/json.
cmake_minimum_required(VERSION 3.24)
project(urnetwork_cpp_provider LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(urnetwork-sdk CONFIG REQUIRED)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(provider main.cpp)
target_link_libraries(provider PRIVATE urnetwork::sdk nlohmann_json::nlohmann_json Threads::Threads)
# the self-test without the sdk library: only the sdk header's data types
add_executable(provider-self-test selftest_main.cpp)
target_include_directories(provider-self-test SYSTEM PRIVATE $<TARGET_PROPERTY:urnetwork::sdk,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(provider-self-test PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
enable_testing()
add_test(NAME provider-self-test COMMAND provider-self-test)
if(MSVC)
  target_compile_options(provider PRIVATE /utf-8 /bigobj)
  target_compile_options(provider-self-test PRIVATE /utf-8 /bigobj)
endif()
if(WIN32)
  add_custom_command(TARGET provider POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:urnetwork::sdk> $<TARGET_FILE_DIR:provider>)
endif()
