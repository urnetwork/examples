cmake_minimum_required(VERSION 3.24)
project(urnetwork_c_messages LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(urnetwork-sdk CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(messages-example main.c runtime.c)
target_link_libraries(messages-example PRIVATE urnetwork::sdk Threads::Threads)
if(WIN32)
  target_link_libraries(messages-example PRIVATE bcrypt)
  add_custom_command(TARGET messages-example POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:urnetwork::sdk> $<TARGET_FILE_DIR:messages-example>)
endif()
