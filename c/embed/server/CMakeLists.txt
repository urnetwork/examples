cmake_minimum_required(VERSION 3.18)
project(urnetwork_c_embed_server LANGUAGES C)
find_package(CURL REQUIRED)
find_package(json-c CONFIG REQUIRED)
add_executable(embed-server embed_server.c)
target_compile_features(embed-server PRIVATE c_std_11)
target_link_libraries(embed-server PRIVATE CURL::libcurl json-c::json-c)
enable_testing()
add_test(NAME embed-server-self-test COMMAND embed-server --self-test)
