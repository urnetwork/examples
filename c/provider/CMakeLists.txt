# Builds the C provider example and its self-test, which does not link the
# SDK library, with the SDK package's urnetwork::sdk target.
cmake_minimum_required(VERSION 3.24)
project(urnetwork_c_provider LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(urnetwork-sdk CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(provider main.c session.c status.c state.c json.c selftest.c)
target_link_libraries(provider PRIVATE urnetwork::sdk Threads::Threads)
# the self-test without the sdk library: only the sdk header's constants
add_executable(provider-self-test selftest_main.c status.c state.c json.c selftest.c)
target_include_directories(provider-self-test PRIVATE $<TARGET_PROPERTY:urnetwork::sdk,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(provider-self-test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME provider-self-test COMMAND provider-self-test)
if(MSVC)
  target_compile_definitions(provider PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(provider-self-test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  target_link_libraries(provider PRIVATE bcrypt)
  target_link_libraries(provider-self-test PRIVATE bcrypt)
  add_custom_command(TARGET provider POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:urnetwork::sdk> $<TARGET_FILE_DIR:provider>)
endif()
