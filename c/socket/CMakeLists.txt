cmake_minimum_required(VERSION 3.24)
project(urnetwork_c_socket LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(urnetwork-sdk CONFIG REQUIRED)
find_package(CURL REQUIRED)
add_executable(socket-example main.c)
target_link_libraries(socket-example PRIVATE urnetwork::sdk CURL::libcurl)
if(WIN32)
  add_custom_command(TARGET socket-example POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:urnetwork::sdk> $<TARGET_FILE_DIR:socket-example>)
endif()
